Միքիայի մարգարեությունը


1 1 Տիրոջ խոսքը, որ Հուդայի թագավորների՝ Հովաթամի, Աքազի ու Եզեկիայի օրերում ամորստացի Միքիային եղավ և Սամարիայի ու Երուսաղեմի մասին բնրան հայտնվեց։ 2գ Լսե՛ք, ո՛վ բոլոր ժողովուրդներ, ուշադրությո՛ւն դարձրու, ո՛վ երկիր ու նրա *բոլոր բնակիչներ, և Տեր Եհովան դձեր դեմ վկա լինի՝ Տերն Ինքը եԻր սուրբ տաճարից, 3 քանի որ ահա զՏերն Իր էտեղից ելնում է և պիտի իջնի ու ըերկրի բարձր տեղերի վրա պիտի պտտվի։ 4 Ու թլեռները Նրա տակ պիտի հալվեն, և հովիտները պիտի պատռվեն կրակի առջև հալած մոմի պես ու զառիվայրից թափված ջրի պես։ 5 Այս ամենը Հակոբի հանցանքի համար ու Իսրայելի տան մեղքերի համար է: Հակոբի հանցանքի պատճառն ո՞վ է, Սամարիան չէ՞: Եվ Հուդայի բարձր տեղերի պատճառն ո՞վ է, Երուսաղեմը չէ՞։ 6 Ուստի Սամարիան ժդաշտի քարակույտ ու այգի տնկելու տեղ պիտի դարձնեմ, նրա քարերը ձորը պիտի գլորեմ ու ինրա հիմքերը պիտի բացեմ։ 7 Եվ նրա բոլոր կուռքերը պիտի կոտրատվեն, ու նրա բոլոր նվիրված բաները կրակով պիտի այրվեն, և նրա բոլոր արձանները բնաջինջ պիտի անեմ, քանի որ դրանք բոզի վարձով է ժողովել, ու բոզի վարձ պիտի դառնան։ 8 Սրա համար լպիտի ողբամ ու պիտի հառաչեմ, խանծածկույթ ու մերկ պիտի շրջեմ, ծչագալների պես կոծ ու ջայլամների պես սուգ պիտի անեմ, 9 քանի որ նրա վերքերն անբժշկելի են, քանի որ կմինչև Հուդա եկավ, հասավ մինչև Իմ ժողովրդի դուռը՝ մինչև Երուսաղեմ։ 10հ Գեթի մեջ մի՛ պատմեք ու բնավ մի՛ լացեք, *Բեթափրայում ձփոշու մեջ թավալվի՛ր: 11 Ո՛վ *Սափիրի բնակիչ, քո ղմերկությունը բաց ունենալով՝ անցի՛ր. **Սաանանի բնակիչը դուրս չի ելնում, Բեթեսելի կոծը իր օթևանը ձեզնից պիտի վերցնի։ 12 Որովհետև *Մարոթի բնակիչը բարիքի համար վշտանում է, քանի որ մինչև Երուսաղեմի դուռը ճՏիրոջից չարիք պիտի իջնի։ 13 Երիվարը կառքի՛ն լծիր, ո՛վ մԼաքիսի բնակիչ. Սիոնի աղջկա մեղքի սկիզբը դա է, քանի որ քո մեջ են Իսրայելի հանցանքները գտնվում։ 14 Դրա համար Մորեսեթ–գեթին ընծաներ պիտի տաս: յ*Աքզիբի տները Իսրայելի թագավորներին պիտի ստեն։ 15 Դեռ քեզ համար ժառանգ պիտի բերեմ, ո՛վ ն*Մարեսայի բնակիչ, Իսրայելի փառքը մինչև շՈդողոմ պիտի հասնի։ 16 Քո ոսիրելի որդիների համար չմազերդ փետի՛ր ու կտրի՛ր, ճաղատությունդ արծվի պես շատացրո՛ւ, քանի որ քեզնից գերության գնացին։ 2 1 Վա՜յ նրանց, որ աանօրինություն են խորհում ու բիրենց անկողինների վրա չարություն են հնարում. նրանք առավոտյան լույսի ժամանակ այն գործի են դնում, քանի որ գնրանց ձեռքերում ուժ կա։ 2 Ու նրանք դագարակներ են ցանկանում ու դրանք հափշտակում են, և տներ ու դրանք առնում են: Եվ մարդուն ու նրա տանը, մարդուն ու նրա ժառանգությանը զրկանք են անում։ 3 Դրա համար Տերն այսպես է ասում. «Ահա Ես եայս ազգի վրա չարիք եմ խորհում, որտեղից չեք կարողանա ձեր պարանոցները հանել ու չեք կարողանա հպարտությամբ քայլել, զորովհետև ժամանակը չար է։ 4 Այն օրը ձեզ վրա էառակ են խոսելու ու տրտմալի ողբով ըողբալու են՝ ասելով. «Ամբողջովին հափշտակվեցինք. իմ ժողովրդի բաժինը փոխեց, Նա ինչպե՜ս է ինձնից վերցնում, մեր ագարակները հետ տալու փոխարեն բաժանում է»։ 5 Դրա համար Տիրոջ ժողովում թվիճակով լար քաշող չպիտի ունենաս։ 6 Մարգարեություն անողներին ասում են. «ժՄի՛ մարգարեացեք»: Նրանք մարգարեություն չպիտի անեն, քանի որ նախատինքը չի վերանա։ 7 Ո՛վ Հակոբի տուն ասվող, մի՞թե Տիրոջ Հոգին կարճացել է, Նրա գործերը սրա՞նք են: Մի՞թե ուղղությամբ քայլողի համար Իմ խոսքերը լավ չեն։ 8 Եվ իմ ժողովուրդը վաղուց է ոտքի ելել թշնամի լինելու համար. ապահովությամբ անցնողների, պատերազմից հետ դարձողների հանդերձների *վրայից պատմուճանը կողոպտում եք։ 9 Իմ ժողովրդի կանանց իրենց զվարճալի տներից վռնդում եք, նրանց երեխաներից հավիտյան Իմ փառքը վերցնում եք։ 10 Ելե՛ք ու գնացե՛ք, քանի որ սա իհանգստության տեղ չէ. լպիղծ լինելու պատճառով մեծ կործանումով բնաջինջ պիտի լինի։ 11 Եթե մի մարդ խքամու և ունայնության հետևից գնա ու ստություն խոսի՝ ասելով՝ «Գինու և օղու համար քեզ մարգարեություն անեմ», նա կլինի այս ժողովրդի մարգարեն։ 12ծ Ո՛վ Հակոբ, բոլորիդ անշուշտ պիտի ժողովեմ, Իսրայելի մնացորդին պիտի հավաքեմ կու Բոսրայի ոչխարների պես իրենց փարախի մեջ եղող հոտերի պես մեկտեղ պիտի դնեմ. հմարդկանց բազմության համար աղմուկ պիտի լինի։ 13 Խրամատություն անողը նրանց առջև է ելնում, նրանք խրամատում են ու դռնով անցնում և դրանով ելնում: Ու ձնրանց թագավորը նրանց առջևից անցնում է, ղև Տերը՝ իբրև նրանց գլուխ, առջևից է գնում»։ 3 1 Ու ասացի. «Հիմա լսե՛ք, ո՛վ Հակոբի գլխավորներ ու Իսրայելի տան իշխաններ, ապետք չէ՞, որ դուք իրավունքը ճանաչեք: 2 Դուք, որ բարությունն ատում եք ու չարությունը սիրում, *մարդկանց վրայից կաշին ու նրանց ոսկորներից միսը հանում եք։ 3 Ու նրանք, որ բԻմ ժողովրդի միսն ուտում են ու նրանց վրայից կաշին հանում ու նրանց ոսկորները փշրում և նրանց պտուկի մեջ ու գկաթսայի մեջ եփելու մսի պես կտոր–կտոր են անում», 4 այն ժամանակ դՏիրոջը կաղաղակեն, բայց նրանց պատասխան չպիտի տա, ու այն ժամանակ Իր երեսը նրանցից կծածկի, որովհետև նրանք չար գործեր են արել։ 5 Այսպես է ասում Տերը եայն մարգարեների համար, որոնք Իմ ժողովրդին մոլորեցնում են ու զիրենց ատամներով կծում են և խաղաղություն են քարոզում, ու էով որ նրանց բերանը բան չդնի, նրա դեմ պատերազմ են հայտարարում: 6ը Դրա համար ձեզ գիշեր կլինի, որ տեսիլք չտեսնեք, և ձեզ խավար կլինի, որ գուշակություն չանեք, թու մարգարեների վրա արևը պիտի մտնի, և նրանց վրա օրը պիտի խավարի։ 7 Ու տեսանողները պիտի ամաչեն, և գուշակողները ամոթով պիտի մնան, բոլորն էլ իրենց շրթունքները պիտի ծածկեն, ժորովհետև Աստծուց պատասխան չպիտի լինի։ 8 Սակայն ես Տիրոջ Հոգով՝ զորությամբ, դատաստանով ու արիությամբ եմ լցված, իոր Հակոբին իր հանցանքը ու Իսրայելին իր մեղքը հայտնեմ։ 9 Հիմա սա լսե՛ք, ո՛վ Հակոբի տան գլխավորներ ու Իսրայելի տան իշխաններ, դուք, որ իրավունքից զզվում եք ու ամեն ուղղություն ծռում եք: 10լ Սիոնն խարյուններով ու Երուսաղեմն անիրավությամբ եք շինում։ 11ծ Նրա գլխավորները կաշառքով են դատում, ու կնրա քահանաները վարձով են սովորեցնում, և նրա մարգարեները փողով գուշակություն են անում հու Տիրոջ վրա են կռթնում՝ ասելով. «Չէ՞ որ Տերը մեր մեջ է, մեզ վրա չարիք չի գա»։ 12 Ուրեմն ձեր պատճառով Սիոնը ձագարակի պես պիտի հերկվի, ու ղԵրուսաղեմը՝ ավերակների դեզեր, ու ճՏիրոջ տան լեռը անտառի բարձր տեղերի պես պիտի լինի։ 4 1 Բայց ավերջին օրերում պիտի լինի, որ Տիրոջ տան լեռը լեռների գլխի վրա պիտի հաստատվի և բլուրներից վեր պիտի բարձրանա, ու ժողովուրդները հեղեղի պես նրան պիտի դիմեն: 2 Եվ շատ ազգեր պիտի գնան ու ասեն. «Եկե՛ք ելնենք Եհովայի լեռը՝ Հակոբի Աստծո տունը, որ մեզ Իր ճանապարհները սովորեցնի, ու Նրա շավիղների մեջ քայլենք, որովհետև օրենքը՝ Սիոնից, ու Տիրոջ խոսքը Երուսաղեմից պիտի ելնի»։ 3 Եվ Նա շատ ժողովուրդների մեջ դատաստան պիտի անի ու զորավոր ազգերին մինչև հեռավոր տեղերը պիտի հանդիմանի, ու բնրանք իրենց սրերը խոփեր ու իրենց նիզակները հոտոցներ պիտի շինեն, ազգ ազգի վրա սուր չպիտի բարձրացնի, գև է՛լ պատերազմ չպիտի սովորեն։ 4դ Այլ ամեն մարդ իր որթի տակ ու իր թզենու տակ պիտի նստի, ու վախեցնող չպիտի լինի, որովհետև Զորաց Տիրոջ բերանն է խոսել։ 5 Քանի որ եբոլոր ժողովուրդներից ամեն մեկն իր աստծո անունով է քայլում, բայց մենք զհավիտյանս հավիտենից մեր Տեր Աստծո անունով պիտի քայլենք։ 6 «Այն օրը,– ասում է Տերը,– էկաղին պիտի ժողովեմ ըու աքսորվածին պիտի հավաքեմ և նրան էլ, որին նեղել եմ։ 7 Եվ կաղին թմնացորդ պիտի անեմ ու հեռու վռնդվածին՝ զորավոր ազգ»: Եվ Տերը Սիոն լեռում ժնրանց վրա պիտի թագավորի այժմվանից մինչև հավիտյան։ 8 Ու դու, ո՛վ հոտի աշտարակ՝ Սիոնի աղջկա բլուր, մինչև քեզ է գալու, քեզ է հասնելու առաջվա իշխանությունը, ու Երուսաղեմի աղջկան՝ թագավորությունը։ 9 Հիմա ինչո՞ւ ես բարձրաձայն գոռում, իմի՞թե քո մեջ թագավոր չկա, կամ թե քո խորհրդականը կորե՞լ է, որ քեզ լծննդկան կնոջ պես ցավ է բռնել։ 10 Ցա՛վ քաշիր ու ծննդկանի պես երկունքի մեջ եղի՛ր, ո՛վ Սիոնի աղջիկ, քանի որ հիմա քաղաքից ելնելու ես ու դաշտում պիտի բնակվես ու մինչև Բաբելոն պիտի գնաս ու այնտեղ պիտի ազատվես. այնտեղ Տերը պիտի ազատի քեզ քո թշնամիների ձեռքից։ 11խ Ու հիմա քեզ վրա շատ ազգեր են ժողովվել, որոնք ասում են. «Թող նա պղծվի, ու մեր աչքերը ծՍիոնը տեսնեն»։ 12 Բայց նրանք կՏիրոջ խորհուրդները չգիտեն ու Նրա դիտավորությունը չեն հասկանում, քանի որ Նա նրանց ժողովելու է, հինչպես որաներն են կալի մեջ ժողովում։ 13ձ Ելի՛ր ու կալը ծեծի՛ր, ո՛վ Սիոնի աղջիկ, քանի որ քո եղջյուրը երկաթից պիտի անեմ և քո կճղակները պղնձից պիտի անեմ, ու ղշատ ժողովուրդների պիտի մանրես: Ու նրանց վաստակը Եհովային ճու նրանց ստացվածքը մամբողջ երկրի Տիրոջը պիտի նվիրեմ։ 5 1 Հիմա, ո՛վ գնդի աղջիկ, քեզ համար գո՛ւնդ կազմիր, մեզ վրա պաշարում է դրվել, աԻսրայելի դատավորի ծնոտին գավազանով պիտի զարկեն։ 2 Եվ դու, ո՛վ բԲեթլեհեմ Եփրաթա, թեև Հուդայի գհազարավորների մեջ փոքր ես, բայց քեզնից պիտի ելնի Ինձ համար Նա, որ Իսրայելի վրա դիշխան պիտի լինի, եու Նրա ելքը հնուց՝ հավիտենական օրերից է։ 3 Դրա համար նրանց ձեռք պիտի տա մինչև զծնողի ծնելու ժամանակը, և էՆրա եղբայրների մնացորդը Իսրայելի որդիների մոտ պիտի վերադառնա։ 4 Ու Նա պիտի կանգնի ու Տիրոջ զորությունով և Իր Տեր Աստծո անվան մեծավայելչությունով ըպիտի հովվի, ու նրանք պիտի բնակվեն, քանի որ հիմա թմինչև երկրի ծայրերը պիտի մեծանա։ 5 Եվ Նա ժխաղաղություն պիտի լինի, երբ ասորեստանցին մեր երկիրը մտնի ու մեր պալատները ոտք դնի, նրա դեմ յոթ հովիվ և ութ իշխանավոր մարդիկ կհանենք։ 6 Ու նրանք Ասորեստանի երկիրը սրով ու իՆեբրովթի երկիրը իր դռների մեջ պիտի սպառեն, և Նա լմեզ պիտի ազատի ասորեստանցիներից, երբ նրանք մեր երկիրը մտնեն ու մեր սահմանների մեջ ոտք դնեն։ 7 Եվ խՀակոբի մնացորդը շատ ժողովուրդների մեջ ծցողի պես պիտի լինի, որ Տիրոջից է գալիս, և շաղի պես, որ կանաչ խոտի վրա է իջնում, որ մարդու չի սպասում ու մարդու որդիների վրա հույսը չի դնում։ 8 Եվ Հակոբի մնացորդը ազգերի մեջ, շատ ժողովուրդների մեջ այնպես պիտի լինի, ինչպես առյուծը՝ անտառի գազանների մեջ, կամ առույգ առյուծը՝ ոչխարների հոտերի մեջ, որ եթե անցնի, կկոխկռտի ու կպատառի, և ազատող չի լինի։ 9 Քո ձեռքը քո հակառակորդների վրա պիտի բարձրանա, ու բոլոր թշնամիներդ պիտի կոտորվեն։ 10կ «Եվ պիտի լինի այն օրը,– ասում է Տերը,– քո ձիերը քո միջից բնաջինջ պիտի անեմ ու քո կառքերը պիտի կորցնեմ։ 11 Եվ քո երկրի քաղաքները բնաջինջ պիտի անեմ ու բոլոր պարիսպներդ պիտի քանդեմ։ 12 Եվ քո ձեռքից կախարդությունները բնաջինջ պիտի անեմ, ու է՛լ հգուշակություն անողներ չպիտի ունենաս։ 13ձ Եվ քո միջից քո կուռքերն ու արձանները բնաջինջ պիտի անեմ, ու ղայլևս քո ձեռքերի գործերին երկրպագություն չես անի։ 14 Եվ քո միջից Աստարովթի կուռքերդ պիտի կործանեմ ու քո քաղաքները պիտի ավերեմ: 15 Եվ Ինձ չլսող ազգերից ճսրտմտությամբ ու բարկությամբ վրեժ պիտի առնեմ»։ 6 1 Հիմա Տիրոջ ասածը լսե՛ք. «Ելի՛ր ու լեռների առջև դա՛տ վարիր, և զարկելով, ձհիվանդացնում եմ և քո մեղքերի համար քեզ ավերակ եմ անում։ 14ղ Դու պիտի ուտես ու չպիտի կշտանաս, և քո անոթությունը փորիդ մեջ պիտի լինի, կհափշտակես, բայց չես կարողանա ազատել, ու ազատածդ սրի պիտի մատնեմ։ 15 Դու ճպիտի սերմես, բայց չպիտի հնձես, դու ձիթապտուղ պիտի տրորես, բայց յուղով չպիտի օծվես: Քաղցու պիտի պատրաստես, բայց գինի չպիտի խմես։ 16 Քանի որ մԱմրիի հրամանները յպահվում են, և նԱքաաբի տան բոլոր գործերը, և դուք նրանց խորհուրդներով եք քայլում, որպեսզի քեզ շամայի անեմ, և *քո բնակիչներին՝ սուլելու ենթակա: Ուստի ոԻմ ժողովրդի նախատինքը պիտի կրեք»։ 7 1 Վա՜յ ինձ, որ ամառային պտղաքաղի պես, այգեկութից ամնացած ճիռերի պես եմ դարձել. ուտելու ողկույզ չկա, բհոգիս կանխահաս պտուղներն է ցանկանում։ 2գ Բարեպաշտը երկրից կորչում է, ու մարդկանց մեջ ուղիղ մեկը չկա. ամենքը արյունների համար դարան են գործում, դամեն մարդ իր եղբորը որոգայթով որսում է։ 3 Ձեռքերով շատ չարություն անելու համար եիշխանը պարգև է պահանջում, զև դատավորը կաշառքի է նայում, ու մեծ մարդն իր սրտի ցանկության համեմատ է խոսում. այսպես նրան ծռում են: 4 Նրանց լավագույնը՝ էփշի պես, ուղիղը փշեղեն ցանկից վատ է: Քո դետերի օրը քո այցելությունը եկավ, նրանց շփոթությունը հիմա պիտի լինի։ 5ը Բարեկամի մի՛ հավատացեք, մտերիմ բարեկամի մի՛ ապավինեք, ծոցդ պառկողից բերանիդ դռները պահի՛ր։ 6 Քանի որ թորդին հորն անարգում է, աղջիկն՝ իր մոր վրա, ու հարսն իր սկեսուրի վրա է ելնում, մարդու թշնամիներն իր ընտանիքն է։ 7 Բայց ժես Տիրոջը կսպասեմ, իմ փրկության Աստծուն կհուսամ. Իմ Աստվածն ինձ պիտի լսի։ 8 Ո՛վ իմ թշնամի, իվրաս մի՛ խնդա. լթեև ընկնեմ, կելնեմ, թեև խավարի մեջ նստեմ, խՏերն ինձ լույս կլինի։ 9ծ Տիրոջ բարկությունը պիտի կրեմ (քանի որ Նրա դեմ մեղք եմ գործել,) մինչև որ Նա իմ դատը պաշտպանի ու ինձ իրավունք անի, կինձ լույսի մեջ հանի, ու Նրա արդարությունը տեսնեմ։ 10 Այն ժամանակ իմ թշնամին պիտի տեսնի, համոթը պիտի ծածկի նրան, որ ինձ ասում էր, թե՝ «ձՈ՞ւր է քո Տեր Աստվածը»: ղԻմ աչքերը նրան կտեսնեն. հիմա նա ճփողոցների ցեխի պես ոտնակոխ պիտի լինի։ 11 Այն օրը, որ քո մպարիսպները կշինվեն, նույն օրը պատվերը հեռացած կլինի։ 12 Այն օրը յԱսորեստանից և *պարսպավոր քաղաքներից ու **պարսպից մինչև Գետը և ծովից ծով ու լեռից լեռ քեզ մոտ կգան։ 13 Սակայն երկիրն ամայի պիտի լինի իր նբնակիչների *պատճառով, որ իրենց գործերի պտղից ուտեն։ 14 Քո գավազանով շհովվի՛ր քո ժողովրդին՝ քո ժառանգության խաշինքը, որ առանձին ոանտառում նստում է Կարմեղոսի մեջ. նրանք առաջվա օրերի պես Բասանի ու Գաղաադի մեջ թող արածեն։ 15 Նրան հրաշքներ պիտի ցույց տամ, չինչպես այն օրերին, որ Եգիպտոսի երկրից ելար։ 16 Ազգերը պպիտի տեսնեն ու իրենց ամբողջ քաջությունից պիտի ամաչեն, ջբերանի վրա ձեռք պիտի դնեն, ու նրանց ականջները պիտի խլանան։ 17 Օձի պես ռհող պիտի լիզեն, երկրի սողունների պես սիրենց թաքնված տեղերից պիտի դողան, վմեր Տեր Աստծուց տպիտի զարհուրեն ու Քեզնից պիտի վախենան։ 18 Քեզ պես Աստված ո՞վ կա, որ րանօրինությունը ներում է ու Իր ժառանգության ցմնացորդի հանցանքն անտեսում է, ուԻր բարկությունը հավիտյան չի պահում, քանի որ ողորմություն է սիրում։ 19 Նա պիտի դառնա, մեզ ողորմություն պիտի անի, մեր անօրինությունները ոտքի տակ պիտի առնի: Այո՛, նրանց բոլոր մեղքերը ծովի խորության մեջ պիտի նետես։ 20փ Հակոբին ճշմարտություն պիտի անես և Աբրահամին՝ ողորմություն, ինչպես առաջվա օրերից ի վեր քմեր հայրերին երդումով խոստացել ես։
